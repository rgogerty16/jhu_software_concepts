# Combined Python dependencies for every module in this repository.
#
# EN.605.256 Modern Software Concepts in Python, Ryan Gogerty.
#
# This is the union of all fourteen per-module requirements files, so a single
# `pip install -r requirements.txt` prepares an environment able to run any
# module in the repository. Each module also keeps its own requirements.txt with
# just its own dependencies, which is the file to use when running one module in
# isolation.
#
# Where two modules pinned the same package differently, the higher floor is
# taken and the conflict is noted inline. Pins were relaxed to `>=` because
# holding thirteen modules to one exact version is what created the conflicts in
# the first place.
#
# Two dependencies cannot be installed by pip and are listed here for
# completeness only:
#   * Graphviz (the `dot` binary), for Module 5's pydeps dependency graph.
#     macOS: brew install graphviz
#   * The Snyk CLI, for Module 5's supply-chain scan.
#     npm install -g snyk
#
# Python 3.10 or newer is required. The committed runs used 3.12.13.

# ── Web application: Modules 1, 3, 4, 5, 6, 13 ──────────────────────────────
# Conflict: Module 1 and Module 6 pinned flask==3.1.0, Modules 3 and 4 pinned
# ==3.1.1, Module 5 pinned ==3.1.3, Module 2's LLM host asked for >=2.3,<4, and
# Module 13 asked for >=3.1. Capitalization also varied between `Flask` and
# `flask`; PyPI treats them as one package, normalized here to lowercase.
flask>=3.1.3

# Postgres driver. Modules 3 through 6 pinned ==3.3.4, Module 13 asked for >=3.2.
psycopg[binary]>=3.3.4

# Message broker client for Module 6's publisher and worker.
pika>=1.3.2

# ── Scraping: Module 2, reused by Modules 3, 4, 5 ───────────────────────────
beautifulsoup4>=4.12.3
selenium>=4.27.1
webdriver-manager>=4.0.2
urllib3>=2.2.3
# Conflict worth flagging: Modules 2, 3, and 4 pinned lxml==5.3.0 while Module 5
# moved to ==6.1.0, a major version apart. Resolved upward. lxml is never
# imported directly; BeautifulSoup selects it by name as its parser backend,
# which is why it does not appear in an import scan.
lxml>=6.1.0

# ── Local language model host: Module 2 ─────────────────────────────────────
# Module 2 cleans scraped program and university names with a TinyLlama GGUF
# model served locally, so no external API is involved.
huggingface_hub>=0.23.0
llama-cpp-python>=0.2.90,<0.3.0

# ── Cloud: Modules 7, 8 ─────────────────────────────────────────────────────
# The AWS SDK, for pulling the dataset from S3 into a SageMaker notebook and
# writing the cleaned dataset back.
boto3>=1.34

# ── Data handling, statistics, and plotting: Modules 8 to 13 ────────────────
# Conflicts resolved upward: numpy >=1.24 against >=1.26, pandas >=2.0 against
# >=2.2, matplotlib >=3.7 against >=3.9.
numpy>=1.26
pandas>=2.2
scipy>=1.10
matplotlib>=3.9

# ── Dashboards: Module 10 ───────────────────────────────────────────────────
seaborn>=0.13
plotly>=5.20

# ── Machine learning: Modules 9, 11, 12, 13 ─────────────────────────────────
scikit-learn>=1.4
joblib>=1.3
torch>=2.5
transformers>=4.46
safetensors>=0.4

# ── Experiment tracking: Module 11 ──────────────────────────────────────────
mlflow>=3.0
# Weights and Biases backed the optional --tracker wandb path.
wandb>=0.17

# ── Report generation: Modules 3, 4, 5, 6, 12, 13 ───────────────────────────
# fpdf2 renders the limitations, report, and write-up PDFs. It was imported by
# Modules 3, 4, 5, and 6 but declared by none of them, so those PDFs could not
# be rebuilt from a clean checkout. Fixed in each module's own requirements.txt

pylint>=3.3.7
